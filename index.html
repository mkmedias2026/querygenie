// Natural language prompt input:
"Find top 10 users with unfulfilled orders placed in the last 30 days"
SELECT u.id, u.email, COUNT(o.id) AS pending_orders
FROM users u JOIN orders o ON u.id = o.user_id
WHERE o.status = 'unfulfilled' AND o.created_at >= NOW() - INTERVAL '30 days'
GROUP BY u.id ORDER BY pending_orders DESC LIMIT 10;
